        "main.cpp"

        "door_control.c"
        "history.c"
        "notifier.c"
        "ha_mqtt.c"
        "captive_dns.c"

        "relay.c"
        "magnet.cpp"
        "wifi_manager.cpp"
        "web_server.cpp"
        "web_assets.cpp"
        "web_help.cpp"          # generated by tools/gen_help.py
        "influx_uploader.cpp"
        "update_check.c"
    INCLUDE_DIRS "."
    REQUIRES
        esp_event
        esp_hw_support          # esp_random, esp_read_mac
        esp_http_client
        esp_http_server
        esp_netif
        esp_partition           # history ring buffer
        esp_system
        esp_timer
        esp_wifi
        nvs_flash
        app_update              # OTA
        json                    # cJSON
        lwip
        mbedtls                 # TLS cert bundle for Influx / notifications
        mdns
        mqtt                    # Home Assistant integration
        led_strip
        esp_adc                 # analogue door sensor
        esp_driver_gpio
        esp_driver_rmt          # SK6812 status LED
        esp_driver_tsens        # on-chip temperature
)
